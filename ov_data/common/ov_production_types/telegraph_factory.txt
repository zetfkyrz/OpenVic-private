# estimate, see docs/WAR_ASSUMPTIONS.md. It is available to a country that has the key of its good (building:telegraph_factory).

	input_goods = {

		steel = 1.0
		glass = 0.5
	}
	output_goods = telegraph
	value = 3
	bonus = {
		trigger = { has_building = machine_parts_factory }

	}
}
